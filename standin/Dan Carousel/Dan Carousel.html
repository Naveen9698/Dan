Script


const root = document.getElementById('dan-dots')
const dan = initDanBase(root)
const dotsContainer = root.querySelector('.dan__dots')
const dots = []

dan.scrollSnapList().forEach((_, index) => {
  const dot = document.createElement('button')
  dot.addEventListener('click', () => dan.scrollTo(index))
  dotsContainer.appendChild(dot)
  dots.push(dot)
})

function updateDots() {
  const index = dan.selectedScrollSnap()
  dots.forEach((dot, i) =>
    dot.classList.toggle('is-active', i === index)
  )
}

updateDots()
dan.on('select', updateDots)

Script


(() => {
  const root = document.getElementById('dan-autoplay-pp')
  const dan = initDanBase(root, { loop: true })

  let autoplayTimer = null

  function playAutoplay() {
    if (!autoplayTimer) {
      autoplayTimer = setInterval(() => {
        dan.scrollNext()
      }, 600)
    }
  }

  function pauseAutoplay() {
    clearInterval(autoplayTimer)
    autoplayTimer = null
  }

  playAutoplay()

  root.querySelector('#autoplay-play').addEventListener('click', playAutoplay)
  root.querySelector('#autoplay-pause').addEventListener('click', pauseAutoplay)
})()
