• Prevents accidental drags (click vs swipe)

• 20–30 → safer for UI with clickable elements

function initDanPrevNext(root, options = {}) {
  if (!root) return null

  const api = DanCarousel(viewport, options)

function initDanKeyboard(root, options = {}) {
  if (!root) return null
  const viewport = root.querySelector('.dan__viewport')
  const api = DanCarousel(viewport, options)

  if (!viewport || !api) return api

  viewport.addEventListener('keydown', function(e) {
    if (e.key === 'ArrowRight') api.scrollNext()
    if (e.key === 'ArrowLeft') api.scrollPrev()
  })

  return api
}

function initDanFade(root, options = {}) {
  if (!root) return null
  const viewport = root.querySelector('.dan__viewport')
  const api = DanCarousel(viewport, options)

  const slides = root.querySelectorAll('.dan__slide')

  function update() {
    const progress = api.scrollProgress()

    slides.forEach((slide, index) => {
      const diff = Math.abs(progress * (slides.length - 1) - index)
      const opacity = 1 - Math.min(diff, 1)
      slide.style.opacity = opacity
    })
  }

  api.on('scroll', update)
  api.on('init', update)

  return api
}

function initDanFeature(root, options = {}) {
  if (!root) return null
  const viewport = root.querySelector('.dan__viewport')
  const api = DanCarousel(viewport, options)

  const bar = root.querySelector('.progress__bar')

  function update() {
    const progress = api.scrollProgress()
    bar.style.width = (progress * 100) + '%'
  }

  api.on('scroll', update)
  api.on('reInit', update)
  update()

  return api
}

function initDanFeature(root, options = {}) {
  if (!root) return null
  const viewport = root.querySelector('.dan__viewport')
  const api = DanCarousel(viewport, options)

  const slides = root.querySelectorAll('.dan__slide')

  function updateHeight() {
    const index = api.selectedScrollSnap()
    const activeSlide = slides[index]
    const height = activeSlide.offsetHeight
    viewport.style.height = height + 'px'
  }

  api.on('select', updateHeight)
  api.on('reInit', updateHeight)
  updateHeight()

  return api
}

function initDanFeature(root, options = {}) {
  if (!root) return null
  const viewport = root.querySelector('.dan__viewport')
  const api = DanCarousel(viewport, options)

  const slides = root.querySelectorAll('.dan__slide')

  function updateClasses() {
    const selected = api.selectedScrollSnap()
    const total = slides.length

    slides.forEach(slide => {
      slide.classList.remove('is-selected', 'is-prev', 'is-next')
    })

    slides[selected].classList.add('is-selected')

    if (selected - 1 >= 0) {
      slides[selected - 1].classList.add('is-prev')
    }

    if (selected + 1 < total) {
      slides[selected + 1].classList.add('is-next')
    }
  }

  api.on('select', updateClasses)
  api.on('reInit', updateClasses)
  updateClasses()

  return api
}

  document.getElementById('dan-state'),
  {}
